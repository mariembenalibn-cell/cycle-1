Procédure chercher_position (T : TAB, N : entier, X : réel ; VAR pos : entier)
Variables
    debut, fin, milieu : entier
Début
    debut ← 1
    fin ← N
    pos ← 0  // Initialisation à 0 (élément non trouvé par défaut)
    
    Tant Que (debut ≤ fin) Et (pos = 0) Faire
        milieu ← (debut + fin) div 2
        
        Si T[milieu] = X Alors
            pos ← milieu  // Élément trouvé, on stocke sa position
        Sinon Si T[milieu] < X Alors
            debut ← milieu + 1  // On cherche dans la moitié droite
        Sinon
            fin ← milieu - 1    // On cherche dans la moitié gauche
        Fin Si
    Fin Tant Que
Fin Procedure

Procédure SEARCH (x : entier, T : Tableau[1..100] de entier, N : entier, var TROUVE : booléen, var POS : entier)
Variables
    i : entier
Début
    TROUVE <- FAUX
    POS <- 0
    i <- 1
    
    TantQue (i <= N) Et (TROUVE = FAUX) Faire
        Si T[i] = x Alors
            TROUVE <- VRAI
            POS <- i
        Sinon
            i <- i + 1
        FinSi
    FinTantQue
Fin Procedure

Procédure DELETE (x : entier, var T : Tableau[1..100] de entier, var N : entier, var TROUVE : booléen)
Variables
    i, j : entier
Début
    TROUVE <- FAUX
    i <- 1
    
    { Étape 1 : Recherche de la première occurrence }
    TantQue (i <= N) Et (TROUVE = FAUX) Faire
        Si T[i] = x Alors
            TROUVE <- VRAI
        Sinon
            i <- i + 1
        FinSi
    FinTantQue
    
    { Étape 2 : Décalage à gauche et mise à jour de la taille }
    Si TROUVE = VRAI Alors
        Pour j De i À N - 1 Faire
            T[j] <- T[j + 1]
        FinPour
        N <- N - 1 { Le tableau a maintenant un élément de moins }
    FinSi
Fin

Procédure DELETE-ALL (x : Entier, var T : Tableau[1..100] de entier, var N : entier, var TROUVE : Booléen)
Variables
    i, j : Entier
Début
    TROUVE <- FAUX
    j <- 1  { Indice pour réécrire les éléments conservés }
    
    Pour i De 1 À N Faire
        Si T[i] = x Alors
            TROUVE <- VRAI
        Sinon
            T[j] <- T[i]
            j <- j + 1
        FinSi
    FinPour
    
    Si TROUVE = VRAI Alors
        N <- j - 1  { Nouvelle taille du tableau après suppressions }
    FinSi
Fin

Procédure CONCAT (T1 : Tableau[1..100] de entier, T2 : Tableau[1..100] de entier, var T3 : Tableau[1..100] de entier, N1 : entier, N2 : entier, var N3 : entier)
Variables
    i : entier
Début
    N3 <- N1 + N2 { La taille de T3 est la somme des tailles de T1 et T2 }
    
    { Étape 1 : Copie des éléments de T1 dans T3 }
    Pour i De 1 À N1 Faire
        T3[i] <- T1[i]
    FinPour
    
    { Étape 2 : Copie des éléments de T2 dans T3 juste après ceux de T1 }
    Pour i De 1 À N2 Faire
        T3[N1 + i] <- T2[i]
    FinPour
Fin
